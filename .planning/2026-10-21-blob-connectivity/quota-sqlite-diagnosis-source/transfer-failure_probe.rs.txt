//! 临时诊断探针：仅显式test-support执行保存固定来源分类，最终修复前删除。
use super::materialization::FileProgressCommitError;
use crate::transfer::blob::facade::FetchTerminalFailure;
use crate::transfer::file::FileTransferApplicationError;
use anyhow::Error;
use std::io::Error as IoError;
use std::path::PathBuf;
use std::sync::atomic::{AtomicU64, Ordering};
use uc_core::ports::FileTransferProjectionError;
use uc_core::ports::InboundReceiveCommitError;

static SEQUENCE: AtomicU64 = AtomicU64::new(0);

pub(crate) fn record_failure(source: &Error, file_save: bool) {
    let Some(root) = std::env::var_os("UC_BLOB_RECEIVE_FAILURE_PROBE") else {
        return;
    };
    let file = source
        .chain()
        .find_map(|source| source.downcast_ref::<FileProgressCommitError>());
    let detail = file.map_or(source, |error| &error.0);
    let commit = detail
        .chain()
        .find_map(|source| source.downcast_ref::<InboundReceiveCommitError>());
    let io = detail
        .chain()
        .find_map(|source| source.downcast_ref::<IoError>());
    let literal = detail.downcast_ref::<&'static str>().copied();
    let transfer = detail
        .chain()
        .find_map(|source| source.downcast_ref::<FileTransferApplicationError>());
    let symbols = detail
        .backtrace()
        .to_string()
        .lines()
        .filter_map(|line| {
            let (number, symbol) = line.trim().split_once(": ")?;
            number.parse::<usize>().ok()?;
            Some(symbol.to_owned())
        })
        .collect::<Vec<_>>();
    let report = serde_json::json!({
        "creation_symbols": symbols,
        "fetch_terminal_failure": detail.chain().any(|source| source.is::<FetchTerminalFailure>()),
        "projection_backend": detail.chain().any(|source| source.is::<FileTransferProjectionError>()),
        "transfer_store": matches!(transfer, Some(FileTransferApplicationError::Store(_))),
        "transfer_publish": matches!(transfer, Some(FileTransferApplicationError::Publish(_))),
        "transfer_repository": matches!(transfer, Some(FileTransferApplicationError::Repository(_))),
        "transfer_conflict": matches!(transfer, Some(FileTransferApplicationError::SessionConflict { .. })),
        "transfer_finished": matches!(transfer, Some(FileTransferApplicationError::TransferAlreadyFinished { .. })),
        "transfer_not_started": matches!(transfer, Some(FileTransferApplicationError::TransferNotStarted { .. })),
        "transfer_invalid_history": matches!(transfer, Some(FileTransferApplicationError::InvalidHistory { .. })),
        "transfer_backwards": matches!(transfer, Some(FileTransferApplicationError::ProgressWentBackwards { .. })),
        "file_save": file_save,
        "file_progress_commit": file.is_some(),
        "attempt_state_mismatch": matches!(commit, Some(InboundReceiveCommitError::AttemptStateMismatch)),
        "artifact_record_missing": matches!(commit, Some(InboundReceiveCommitError::ArtifactRecordMissing)),
        "artifact_state_mismatch": matches!(commit, Some(InboundReceiveCommitError::ArtifactStateMismatch)),
        "directory_record_missing": matches!(commit, Some(InboundReceiveCommitError::DirectoryPublishRecordMissing)),
        "backend": matches!(commit, Some(InboundReceiveCommitError::Backend(_))),
        "io_kind": io.map(|io| format!("{:?}", io.kind())),
        "context_materialize": literal == Some("materialize inbound blobs"),
        "context_identity": literal == Some("verify received file set identity"),
        "no_matching_entry": literal == Some("file progress capture returned no matching entry"),
        "missing_original_identity": literal == Some("saved receive item is missing its original identity"),
        "source_chain_length": detail.chain().count(),
    });
    let root = PathBuf::from(root);
    std::fs::create_dir_all(&root).expect("创建临时固定分类诊断目录");
    let seq = SEQUENCE.fetch_add(1, Ordering::Relaxed);
    std::fs::write(
        root.join(format!("failure-{}-{seq}.json", std::process::id())),
        serde_json::to_vec_pretty(&report).expect("编码固定分类"),
    )
    .expect("保存固定分类");
}
